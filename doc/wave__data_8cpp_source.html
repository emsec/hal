1155  mDirty = true;
1156  return;
1157  }
1158  for (int ibit = 0; ibit < nChildren; ibit++)
1159  {
1160  undef |= (1 << ibit);

1173  if (!wd) continue;
1174  for (u64 t : wd->data().keys())
1175  tIndex.insert(t,ibit);
1176  }

1178  for (auto it = tIndex.constBegin(); it != tIndex.constEnd(); ++it)
1179  {

1182  mData.insert(t0, undef ? -1 : value);

1194  value |= mask;

1197  }
