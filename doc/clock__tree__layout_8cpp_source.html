42  // The sizes of the drawing follow from the length of a typical connection, because that decides how large a
43  // node can be before it touches its neighbours. The ratios are those of a clock tree with connections of
44  // about 25 um that is drawn with a node radius of 1 um, a line width of 0.5 um, a picture margin of 20 um and
45  // 2 pixels per um.
46  constexpr double REFERENCE_EDGE_LENGTH = 25.0;
47  constexpr double NODE_RADIUS_RATIO = 1.0 / REFERENCE_EDGE_LENGTH;
48  constexpr double EDGE_WIDTH_RATIO = 0.5 / REFERENCE_EDGE_LENGTH;
49  constexpr double PICTURE_MARGIN_RATIO = 20.0 / REFERENCE_EDGE_LENGTH;
50  constexpr double EDGE_PICTURE_PIXELS = 2.0 * REFERENCE_EDGE_LENGTH;
51 
52  // Without a single connection there is no typical one, a tenth of the extent of the gates takes its place.
53  constexpr double EXTENT_RATIO = 0.1;

57  double min_x;

63  Bounds bounds_of( const Layout &layout )
64  {
65  const Node &first = layout.nodes.front();
66  Bounds bounds = { first.x, first.x, first.y, first.y };
67 
68  for( const Node &node : layout.nodes )
69  {
70  bounds.min_x = std::min( bounds.min_x, node.x );
71  bounds.max_x = std::max( bounds.max_x, node.x );
72  bounds.min_y = std::min( bounds.min_y, node.y );
73  bounds.max_y = std::max( bounds.max_y, node.y );
74  }
75 
76  return bounds;
77  }

79  double typical_edge_length( const Layout &layout )
80  {
81  std::vector<double> lengths;
82  lengths.reserve( layout.edges.size() );
83 
84  for( const Edge &edge : layout.edges )
85  {
86  const Node &from = layout.nodes[edge.from];
87  const Node &to = layout.nodes[edge.to];
88 
89  if( const double length = std::hypot( to.x - from.x, to.y - from.y ); length > 0.0 )
90  {
91  lengths.push_back( length );
92  }
93  }
94 
95  if( !lengths.empty() )
96  {
97  const auto middle = lengths.begin() + lengths.size() / 2;
98  std::nth_element( lengths.begin(), middle, lengths.end() );
99  return *middle;
100  }
101 
102  const Bounds bounds = bounds_of( layout );
103  const double extent = std::max( bounds.max_x - bounds.min_x, bounds.max_y - bounds.min_y );
104 
105  return extent > 0.0 ? EXTENT_RATIO * extent : 1.0;
106  }

114  Result<Layout> build_layout( const cte::ClockTree &tree,
115  const std::optional<std::string> &x_key,
116  const std::optional<std::string> &y_key )
117  {
118  if( x_key.has_value() != y_key.has_value() )
119  {
120  return ERR( "give the parameter names of both coordinates or of none" );
121  }
122 
123  std::vector<const Gate *> gates = tree.get_gates();
124  if( gates.empty() )
125  {
126  return ERR( "the clock tree contains no gates" );
127  }
128 
129  // The clock tree keeps its gates in a hash map, sort them to get the same layout every time.
130  std::sort(
131  gates.begin(), gates.end(), []( const Gate *a, const Gate *b ) { return a->get_id() < b->get_id(); } );
132 
133  Layout layout;
134  std::unordered_map<const Gate *, std::size_t> node_of_gate;
135 
136  for( const Gate *gate : gates )
137  {
138  // Coordinates from gate parameters can be negative, a location in HAL can't.
139  std::optional<double> x;
140  std::optional<double> y;
141  if( x_key.has_value() )
142  {
143  x = cte::read_parameter_coordinate( gate, *x_key );
144  y = cte::read_parameter_coordinate( gate, *y_key );
145  }
146  else if( gate->has_location() )
147  {
148  x = gate->get_location_x();
149  y = gate->get_location_y();
150  }
151 
152  if( !x.has_value() || !y.has_value() )
153  {
154  layout.gates_without_coordinates.push_back( gate->get_id() );
155  continue;
156  }
157 
158  node_of_gate[gate] = layout.nodes.size();
159  layout.nodes.push_back( { gate->get_id(), *x, *y } );
160  }
161 
162  if( layout.nodes.empty() )
163  {
164  const std::string expected = x_key.has_value() ? "the gate parameters '" + *x_key + "' and '" + *y_key + "'"
165  : std::string( "a location in the layout" );
166  return ERR( "none of the " + std::to_string( gates.size() )
167  + " gates of the clock tree has coordinates, expected " + expected );
168  }
169 
170  // Nets are not drawn, so the only connections left are those between gates.
171  for( const Gate *gate : gates )
172  {
173  const auto successors = tree.get_neighbors( gate, IGRAPH_OUT );
174  if( successors.is_error() )
175  {
176  return ERR( "could not read the successors of gate with ID " + std::to_string( gate->get_id() ) + ": "
177  + successors.get_error().get() );
178  }
179 
180  for( const auto &[ptr, type] : successors.get() )
181  {
182  if( type != cte::PtrType::GATE )
183  {
184  continue;
185  }
186 
187  const auto from = node_of_gate.find( gate );
188  const auto to = node_of_gate.find( static_cast<const Gate *>( ptr ) );
189 
190  if( from == node_of_gate.end() || to == node_of_gate.end() )
191  {
192  layout.omitted_edges++;
193  continue;
194  }
195 
196  layout.edges.push_back( { from->second, to->second } );
197  }
198  }
199 
200  return OK( std::move( layout ) );
201  }

203  Result<Style> resolve_style( const Layout &layout, const StyleOverrides &overrides )
204  {
205  if( layout.nodes.empty() )
206  {
207  return ERR( "the layout has no gates to size the drawing after" );
208  }
209  if( !is_valid( overrides.node_radius, false ) )
210  {
211  return ERR( "node_radius must be a positive number" );
212  }
213  if( !is_valid( overrides.edge_width, false ) )
214  {
215  return ERR( "edge_width must be a positive number" );
216  }
217  if( !is_valid( overrides.picture_margin, true ) )
218  {
219  return ERR( "picture_margin must not be negative" );
220  }
221  if( !is_valid( overrides.pixels_per_unit, false ) )
222  {
223  return ERR( "pixels_per_unit must be a positive number" );
224  }
225 
226  const double length = typical_edge_length( layout );
227 
228  Style style;
229  style.node_radius = overrides.node_radius.value_or( NODE_RADIUS_RATIO * length );
230  style.edge_width = overrides.edge_width.value_or( EDGE_WIDTH_RATIO * length );
231  style.picture_margin = overrides.picture_margin.value_or( PICTURE_MARGIN_RATIO * length );
232 
233  // A typical connection gets the same number of pixels in every picture. The scene lowers this for a picture that
234  // would be larger than allowed, which only it can tell.
235  style.pixels_per_unit = overrides.pixels_per_unit.value_or( EDGE_PICTURE_PIXELS / length );
236  style.pixels_per_unit_is_fixed = overrides.pixels_per_unit.has_value();
237 
238  return OK( style );
239  }
