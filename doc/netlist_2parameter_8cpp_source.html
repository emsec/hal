506  {
507  const std::string v = normalize_value(type, value);
508  switch (type)
509  {
510  case Type::Boolean: {
511  return v == "true" || v == "false";
512  }
513  case Type::BitVector: {

517  if (parsed.is_error())
518  {

523  }
524  // size > 64: validate structurally — count significant bits without
525  // converting to u64 (which would overflow for wide values).
526  if (v.size() < 3 || v[0] != '0')
527  {
528  return false;
529  }

532  std::string valid_chars;
533  switch (bc)
534  {
535  case 'b': bpd = 1; valid_chars = "01"; break;
536  case 'o': bpd = 3; valid_chars = "01234567"; break;
537  case 'x': bpd = 4; valid_chars = "0123456789ABCDEF"; break;
538  default: return false;
539  }

541  if (digits.empty())
542  {
543  return false;
544  }

546  {

548  {
549  return false;
550  }
551  }

553  size_t i = 0;
