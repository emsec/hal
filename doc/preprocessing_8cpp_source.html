262  }

266  }

271  if (clk_pins.size() != 1)

466  Result<u32> remove_no_load_wires(Netlist* nl, const std::vector<Net*>& nets)

473  // Vivado names a net that only exists to give an unused output pin something to drive 'NLW_<instance>_<pin>_UNCONNECTED'.
474  // When the hierarchy is flattened, the instance is a path, so the prefix either starts the name or follows a '/';
475  // a multi-bit no load wire is written as a vector and its bits carry a trailing index like '(2)' or '[2]'.
476  const auto is_no_load_wire = [](const Net* n) {
477  const std::string& name = n->get_name();
478  const std::string prefix = "NLW_";
479  const std::string suffix = "_UNCONNECTED";
480 
481  if (name.compare(0, prefix.size(), prefix) != 0 && name.find("/" + prefix) == std::string::npos)
482  {
483  return false;
484  }
485 
486  std::string::size_type end = name.size();
487  if (end > 0 && (name[end - 1] == ')' || name[end - 1] == ']'))
488  {
489  const auto open = name.find_last_of("([");
490  if (open == std::string::npos || open + 1 >= end - 1 || name.find_first_not_of("0123456789", open + 1) != end - 1)
491  {
492  return false;
493  }
494  end = open;
495  }
496 
497  return end > prefix.size() + suffix.size() && name.compare(end - suffix.size(), suffix.size(), suffix) == 0;
498  };
499 
500  std::vector<Net*> to_delete;
501  for (Net* n : nets.empty() ? nl->get_nets() : nets)
502  {
503  if (n == nullptr || n->get_netlist() != nl)
504  {
505  return ERR("cannot remove no load wires: a net of the scope does not belong to netlist with ID " + std::to_string(nl->get_id()));
506  }
507 
508  if (is_no_load_wire(n) && n->get_num_of_destinations() == 0 && !n->is_global_output_net())
509  {
