323  void WaveWidget::visualizeCurrentNetState(double tCursor, int xpos)
324  {
325  QSet<Net*> netState[3]; // x, 0, 1
326 
327  for (auto it = mWaveItemHash->begin(); it != mWaveItemHash->end(); ++it)
328  {
329  WaveItem* wi = it.value();
330  if (!wi) continue;
331  const WaveData* wd = wi->wavedata();
332  if (!wd) continue;
333  Net* n = gNetlist->get_net_by_id(wd->id());
334  if (!n) continue;
335  int val = wi->cursorValue(tCursor, xpos);
336  if (val < -1 || val > 1) continue;
337  netState[val+1].insert(n);
338  }
339 
340  for (int i=0; i<3; i++)
341  {
342  Grouping* grp = gNetlist->get_grouping_by_id(mGroupIds[i]);
343  if (!grp) continue;
344  for (Net* n : grp->get_nets())
345  {
346  auto it = netState[i].find(n);
347  if (it == netState[i].end())
348  grp->remove_net(n);
349  else
350  netState[i].erase(it);
351  }
352  for (Net* n : netState[i])
353  grp->assign_net(n, true);
354  }
