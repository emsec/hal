47  const char *shape_of( const Gate *gate )
48  {
49  if( is_ff( gate ) )
50  {
51  return "box";
52  }
53  if( is_buffer( gate ) || is_inverter( gate ) || is_delay( gate ) )
54  {
55  // points in the direction of the signal, since the graph is laid out from top to bottom
56  return "invtriangle";
57  }
58  if( is_mux( gate ) )
59  {
60  return "diamond";
61  }
62  if( is_oscillator( gate ) || is_pll( gate ) )
63  {
64  return "cylinder";
65  }
66  return "hexagon";
67  }

69  // quotes text as a DOT ID, e.g., names like a[3] or escaped Verilog identifiers
70  std::string quoted( const std::string &text )
71  {
72  std::string result = "\"";
73  for( const char character : text )
74  {
75  if( character == '"' || character == '\\' )
76  {
77  result += '\\';
78  }
79  result += character;
80  }
81  return result + "\"";
82  }

84  // the shortest text that reads back as the same number, e.g., 100 or 12.5, independent of the locale
85  std::string number( const double value )
86  {
87  char text[32];
88  const auto [end, error] = std::to_chars( text, text + sizeof( text ), value );
89  return { text, end };
90  }

93  Result<std::monostate> export_dot( const ClockTree &clock_tree,
94  const std::string &pathname,
95  const std::optional<std::string> &x_key,
96  const std::optional<std::string> &y_key )
97  {
98  if( x_key.has_value() != y_key.has_value() )
99  {
100  return ERR( "couldn't export clock tree to '" + pathname + "': give both coordinate keys or none" );
101  }
102 
103  std::ofstream dot_file( pathname );
104  if( !dot_file )
105  {
106  return ERR( "couldn't export clock tree to '" + pathname + "'" );
107  }
108 
109  dot_file << "digraph {\n"
110  << " comment=\"created by HAL plugin clock_tree_extractor\"\n"
111  << " node [color=black fontcolor=black]\n"
112  << " edge [color=black]\n";
113 
114  const igraph_t *graph = clock_tree.get_igraph();
115  const igraph_int_t vertex_count = igraph_vcount( graph );
116 
117  // the node ID of every vertex, reused for the edges
118  std::vector<std::string> node_ids( vertex_count );
119  u32 gates_without_coordinates = 0;
120 
121  for( igraph_int_t vertex = 0; vertex < vertex_count; vertex++ )
122  {
123  const auto object = clock_tree.get_ptr_from_vertex( vertex );
124  if( object.is_error() )
125  {
126  return ERR_APPEND( object.get_error(), "couldn't export clock tree to '" + pathname + "'" );
127  }
128  const auto [ptr, type] = object.get();
129 
130  if( type == NET )
131  {
132  // prefixed, since a numeric net name would otherwise merge with the gate of that ID
133  const Net *net = static_cast<const Net *>( ptr );
134  node_ids[vertex] = "net_" + std::to_string( net->get_id() );
135  dot_file << " " << node_ids[vertex] << " [label=" << quoted( net->get_name() ) << " shape=circle];\n";
136  continue;
137  }
138 
139  const Gate *gate = static_cast<const Gate *>( ptr );
140  node_ids[vertex] = std::to_string( gate->get_id() );
141  dot_file << " " << node_ids[vertex] << " [instance=" << quoted( gate->get_name() )
142  << " type=" << quoted( gate->get_type()->get_name() ) << " shape=" << shape_of( gate );
143 
144  // buffers, inverters and delays share the triangle, so the label tells them apart
145  if( is_inverter( gate ) )
146  {
147  dot_file << " label=\"¬ " << node_ids[vertex] << "\"";
148  }
149  else if( is_delay( gate ) )
150  {
151  dot_file << " label=\"Δ " << node_ids[vertex] << "\"";
152  }
153 
154  // coordinates from gate parameters can be negative, a location in HAL can't
155  std::optional<double> x;
156  std::optional<double> y;
157  if( x_key.has_value() )
158  {
159  x = read_parameter_coordinate( gate, *x_key );
160  y = read_parameter_coordinate( gate, *y_key );
161  }
162  else if( gate->has_location() )
163  {
164  x = gate->get_location_x();
165  y = gate->get_location_y();
166  }
167 
168  if( x.has_value() && y.has_value() )
169  {
170  dot_file << " x=" << number( *x ) << " y=" << number( *y );
171  }
172  else
173  {
174  gates_without_coordinates++;
175  }
176 
177  dot_file << "];\n";
178  }
179 
180  // every edge, also those that no root reaches, e.g., in a ring oscillator without an enable
181  const igraph_int_t edge_count = igraph_ecount( graph );
182  for( igraph_int_t edge = 0; edge < edge_count; edge++ )
183  {
184  igraph_int_t from;
185  igraph_int_t to;
186  igraph_edge( graph, edge, &from, &to ); // can't fail for an existing edge ID
187  dot_file << " " << node_ids[from] << " -> " << node_ids[to] << ";\n";
188  }
189 
190  dot_file << "}\n";
191  dot_file.close();
192  if( !dot_file )
193  {
194  return ERR( "couldn't write clock tree to '" + pathname + "'" );
195  }
196 
197  if( x_key.has_value() && gates_without_coordinates > 0 )
198  {
199  log_warning( "clock_tree_extractor",
200  "no valid coordinates in parameters '{}' and '{}' for {} gate(s)",
201  *x_key,
202  *y_key,
203  gates_without_coordinates );
204  }
205 
206  return OK( {} );
207  }
