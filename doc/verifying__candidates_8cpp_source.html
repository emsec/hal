736  for (const auto& [c, indices] : count_to_indices)
737  {
738  if (indices.size() > 1)
739  {
740  overfull_bins++;
741  }
742  }
743

745  if (overfull_bins != 1)
746  {

749

751  if (count_to_indices.rbegin()->second.size() == 1)
