96  void ClockTreeGraphicsView::showArea( const QRectF &area )
97  {
98  if( !mScene->hasLayout() )
99  {
100  return;
101  }
102 
103  // Some air around the gates, so they are not at the edge of the view: a tenth of the area, but at least a few
104  // gates wide, which is all there is around a single gate.
105  const double air = std::max( 0.1 * std::max( area.width(), area.height() ),
106  AREA_AIR_NODE_DIAMETERS * 2.0 * mScene->style().node_radius );
107 
108  fitRect( area.adjusted( -air, -air, air, air ) );
109 
110  // Fitting a few gates zooms in further than is of use, bring the zoom back within its limits.
111  zoomBy( 1.0 );
112  centerOn( area.center() );
113  }

115  double ClockTreeGraphicsView::fitScale() const
116  {
117  const QRectF rect = sceneRect();
118  if( rect.isEmpty() )
119  {
120  return 1.0;
121  }
122 
123  return std::min( viewport()->width() / rect.width(), viewport()->height() / rect.height() );
124  }

126  void ClockTreeGraphicsView::zoomBy( double factor )
127  {
128  if( !mScene->hasLayout() )
129  {
130  return;
131  }
132 
133  const double current = transform().m11();
134  const double fit = fitScale();
135 
136  const double minimum = MIN_ZOOM_OF_FIT * fit;
137  const double maximum = std::max( 2.0 * fit, MAX_NODE_DIAMETER_PIXELS / ( 2.0 * mScene->style().node_radius ) );
138 
139  const double target = std::clamp( current * factor, minimum, maximum );
140  if( target == current )
141  {
142  return;
143  }
144 
145  scale( target / current, target / current );
146  }

148  void ClockTreeGraphicsView::wheelEvent( QWheelEvent *event )
149  {
150  // 120 is one notch of a mouse wheel. A trackpad reports many small steps, which zoom correspondingly little.
151  const double notches = event->angleDelta().y() / 120.0;
152  if( notches == 0.0 )
153  {
154  event->ignore();
155  return;
156  }
157 
158  zoomBy( std::pow( ZOOM_STEP, notches ) );
159  event->accept();
160  }
