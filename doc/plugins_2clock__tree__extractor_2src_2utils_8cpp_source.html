94  std::optional<double> read_parameter_coordinate( const Gate *gate, const std::string &name )
95  {
96  const Result<std::string> parameter = gate->get_parameter_value( name );
97  if( parameter.is_error() )
98  {
99  return std::nullopt;
100  }
101 
102  const std::string &value = parameter.get();
103  const std::size_t first = value.find_first_not_of( " \t\r\n" );
104  if( first == std::string::npos )
105  {
106  return std::nullopt;
107  }
108  const std::string text = value.substr( first, value.find_last_not_of( " \t\r\n" ) - first + 1 );
109 
110  // bit vectors are stored in hexadecimal, which not every standard library reads as a double
111  if( text.rfind( "0x", 0 ) == 0 )
112  {
113  u64 coordinate = 0;
114  if( const auto [end, error] = std::from_chars( text.data() + 2, text.data() + text.size(), coordinate, 16 );
115  error != std::errc() || end != text.data() + text.size() )
116  {
117  return std::nullopt;
118  }
119  return static_cast<double>( coordinate );
120  }
121 
122  // read with the classic locale, so a coordinate has a decimal point no matter where the program runs
123  std::istringstream stream( text );
124  stream.imbue( std::locale::classic() );
125 
126  double coordinate = 0.0;
127  stream >> coordinate;
128 
129  // the whole value must be the number, "12 um" is not a coordinate
130  if( stream.fail() || !stream.eof() || !std::isfinite( coordinate ) )
131  {
132  return std::nullopt;
133  }
134  return coordinate;
135  }
