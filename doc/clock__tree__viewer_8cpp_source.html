115  Result<ctv::Layout> extract_layout( const Netlist *netlist,
116  const std::optional<std::string> &x_key,
117  const std::optional<std::string> &y_key )
118  {
119  // An exception escaping the thread would end the program.
120  try
121  {
122  const Result<std::unique_ptr<cte::ClockTree>> tree = cte::ClockTree::from_netlist( netlist );
123  if( tree.is_error() )
124  {
125  return ERR( tree.get_error().get() );
126  }
127 
128  return layout_of( *tree.get(), netlist, x_key, y_key );
129  } catch( const std::exception &e )
130  {
131  return ERR( std::string( "the extraction failed: " ) + e.what() );
132  }
133  }

139  QString list_gate_ids( const std::vector<u32> &ids )
140  {
141  QStringList listed;
142  for( std::size_t idx = 0; idx < std::min( ids.size(), MAX_LISTED_GATES ); idx++ )
143  {
144  listed.push_back( QString::number( ids[idx] ) );
145  }
146 
147  QString result = listed.join( ", " );
148  if( ids.size() > MAX_LISTED_GATES )
149  {
150  result += QString( " and %1 more" ).arg( ids.size() - MAX_LISTED_GATES );
151  }
152  return result;

155  // Runs the work in the GUI thread and waits for it, so a script thread can hand work to a viewer. The viewer must
156  // be looked up inside the work: the GUI may delete it at any time between two of its events.
157  template<typename Work>
158  void run_in_gui_thread( Work &&work )
159  {
160  QCoreApplication *application = QCoreApplication::instance();
161  if( QThread::currentThread() == application->thread() )
162  {
163  work();
164  return;
165  }
166  QMetaObject::invokeMethod( application, std::forward<Work>( work ), Qt::BlockingQueuedConnection );
167  }

303  bool ClockTreeViewer::setClockTree( const cte::ClockTree &tree,
304  const ctv::StyleOverrides &overrides,
305  const std::optional<std::string> &x_key,
306  const std::optional<std::string> &y_key )
307  {
308  // Laying out only reads the clock tree and its gates and touches no widget, so it happens right here. Only
309  // showing the result has to happen in the GUI thread.
310  const Result<ctv::Layout> layout = layout_of( tree, gNetlist, x_key, y_key );
311 
312  bool shown = false;
313  run_in_gui_thread( [&]() {
314  ClockTreeViewer *viewer = open_viewer();
315  if( viewer != nullptr )
316  {
317  shown = viewer->display( layout, overrides, x_key, y_key );
318  }
319  } );
320  return shown;
321  }

323  bool ClockTreeViewer::display( const Result<ctv::Layout> &layout,
324  const ctv::StyleOverrides &overrides,
325  const std::optional<std::string> &x_key,
326  const std::optional<std::string> &y_key )
327  {
328  if( layout.is_error() )
329  {
330  return showError( QString::fromStdString( layout.get_error().get() ) );
331  }
332 
333  const Result<ctv::Style> style = ctv::resolve_style( layout.get(), overrides );
334  if( style.is_error() )
335  {
336  return showError( QString::fromStdString( style.get_error().get() ) );
337  }
338 
339  mOverrides = overrides;
340  mXKey = x_key;
341  mYKey = y_key;
342  mLayout = layout.get();
343 
344  mNodeOfGate.clear();
345  for( std::size_t idx = 0; idx < mLayout.nodes.size(); idx++ )
346  {
347  mNodeOfGate.insert( mLayout.nodes[idx].gate_id, static_cast<int>( idx ) );
348  }
349 
350  // Whatever is selected in HAL when the plot appears is highlighted right away. The highlight is part of the colors,
351  // so they are only worked out after it.
352  highlightFromSelection();
353 
354  mScene->setLayout( mLayout, style.get(), currentColors() );
355  mScene->setHighlight( highlightedNodes() );
356  mShowsError = false;
357  mStack->setCurrentWidget( mView );
358  mView->fitToScene();
359  updateActions();
360 
361  log_info(
362  "clock_tree_viewer", "Showing {} gates and {} connections.", mLayout.nodes.size(), mLayout.edges.size() );
363 
364  if( mLayout.gates_without_coordinates.empty() )
365  {
366  showWarning( QString() );
367  }
368  else
369  {
370  const std::size_t total = mLayout.nodes.size() + mLayout.gates_without_coordinates.size();
371 
372  log_warning(
373  "clock_tree_viewer",
374  "{} of {} gates have no coordinates and are not shown, {} connections are left out. Gate IDs: {}",
375  mLayout.gates_without_coordinates.size(),
376  total,
377  mLayout.omitted_edges,
378  list_gate_ids( mLayout.gates_without_coordinates ).toStdString() );
379 
380  showWarning( QString( "%1 of %2 gates have no coordinates and are not shown (%3 connections left out)" )
381  .arg( mLayout.gates_without_coordinates.size() )
382  .arg( total )
383  .arg( mLayout.omitted_edges ) );
384  }
385 
386  return true;
387  }

389  bool ClockTreeViewer::showError( const QString &message )
390  {
391  log_error( "clock_tree_viewer", "{}", message.toStdString() );
392 
393  // The errors of a Result start with the source location they come from, which says nothing to the user.
394  static const QRegularExpression source_location( R"(\[[^\]\n]*:\d+\]\s*)" );
395 
396  QString text = message;
397  text.remove( source_location );
398 
399  mScene->clearLayout();
400  mLayout = ctv::Layout();
401  mNodeOfGate.clear();
402  mHighlightedGates.clear();
403  mShowsError = true;
404  applyThemeColors();
405  mMessageLabel->setText( "The clock tree cannot be shown:\n\n" + text.trimmed() );
406  mStack->setCurrentWidget( mMessageLabel );
407  showWarning( QString() );
408  updateActions();
409 
410  return false;
411  }

451  QColor ClockTreeViewer::gateColor( u32 gateId, const QColor &fallback ) const
452  {
453  // The same order of precedence as the items of the graph view have, see GraphicsItem::penColor.
454  if( mHighlightedGates.contains( gateId ) )
455  {
456  return HAL_SELECTION_COLOR;
457  }
458 
459  if( gNetlist == nullptr )
460  {
461  return fallback;
462  }
463 
464  const Gate *gate = gNetlist->get_gate_by_id( gateId );
465  if( gate == nullptr )
466  {
467  return fallback;
468  }
469 
470  if( gContentManager != nullptr && gContentManager->getGroupingManagerWidget() != nullptr )
471  {
472  const QColor grouping =
473  gContentManager->getGroupingManagerWidget()->getModel()->colorForItem( ItemType::Gate, gateId );
474  if( grouping.isValid() )
475  {
476  return grouping;
477  }
478  }
479 
480  if( const Module *module = gate->get_module(); module != nullptr && gNetlistRelay != nullptr )
481  {
482  return gNetlistRelay->getModuleColor( module->get_id() );
483  }
484 
485  return fallback;
486  }

571  std::size_t ClockTreeViewer::applyHighlight( const std::vector<u32> &gateIds )
572  {
573  mHighlightedGates.clear();
574 
575  for( const u32 gateId : gateIds )

578  {

580  }
581  }

589  std::size_t ClockTreeViewer::highlightGates( const std::vector<u32> &gateIds )
590  {
591  std::size_t highlighted = 0;
592  bool found = false;
593 
594  run_in_gui_thread( [&]() {
595  ClockTreeViewer *viewer = open_viewer();
596  if( viewer != nullptr )
597  {
598  found = true;
599  highlighted = viewer->applyHighlight( gateIds );
600  }
601  } );
602 
603  if( found && !gateIds.empty() )
604  {
605  log_info( "clock_tree_viewer",
606  "Highlighted {} of {} gates, the others are not in the plot.",
607  highlighted,
608  gateIds.size() );
609  }
610 
611  return highlighted;
612  }

643  void ClockTreeViewer::handleExtractClicked()
644  {
645  if( gNetlist == nullptr )
646  {
647  showError( "No netlist is loaded." );
648  return;
649  }
650 
651  if( mExtraction != nullptr )
652  {
653  return;
654  }

656  // Everything the thread needs is copied now, it must not read the members of the viewer.

694
