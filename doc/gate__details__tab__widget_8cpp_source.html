39  // Returns the hex digits (without "0x") of the INIT parameter of a LUT gate, padded with leading zeros to the
40  // declared width. Falls back to the default of the gate type if the gate holds no instance value.
41  Result<std::string> getLutInitHex(const Gate* gate, const std::string& initKey)
42  {
43  Result<Parameter> decl = gate->has_parameter(initKey) ? gate->get_parameter_declaration(initKey) : gate->get_type()->get_parameter(initKey);
44  if (decl.is_error())
45  {
46  return ERR_APPEND(decl.get_error(), "no INIT parameter '" + initKey + "' declared");
47  }
48 
49  std::string hex = gate->has_parameter(initKey) ? gate->get_parameter_value(initKey).get() : decl.get().get_default_value();
50  if (hex.size() >= 2 && hex[0] == '0' && (hex[1] == 'x' || hex[1] == 'X'))
51  {
52  hex = hex.substr(2);
53  }
54  if (hex.empty())
55  {
56  hex = "0";
57  }
58 
59  const size_t numDigits = (static_cast<size_t>(decl.get().get_size()) + 3) / 4;
60  if (hex.size() < numDigits)
61  {
62  hex.insert(0, numDigits - hex.size(), '0');
63  }
64  return OK(hex);
65  }

322  void GateDetailsTabWidget::handleLutConfigContextMenuRequested(QPoint pos)
323  {
324  // Resolve the INIT parameter and bit slice of the currently selected pin
325  const LUTComponent* lc = mCurrentGate->get_type()->get_component_as<LUTComponent>([](const GateTypeComponent* c) { return LUTComponent::is_class_of(c); });
326  const LUTComponent::LUTOutputConfig* cfg = (lc != nullptr) ? lc->get_output_pin_config(mSelectedLutPin) : nullptr;
327  if (cfg == nullptr)
328  {
329  log_error("gui", "Could not load LUT configuration of pin '{}' from gate with id {}.", mSelectedLutPin, mCurrentGate->get_id());
330  return;
331  }
332  const std::string initKey = cfg->init_identifier;
333  const u32 bitOffset = cfg->bit_offset;
334  const u32 bitCount = cfg->bit_count;
335 
336  QMenu menu;
337 
338  menu.addAction("Configuration string to clipboard", [this]() { QApplication::clipboard()->setText(mLutConfigLabel->text().remove(" 0x")); });
339  menu.addAction("Change configuration string", [this, initKey, bitOffset, bitCount]() {
340  // Strip leading " 0x" from the label to get the current hex digits
341  QString currentHex = mLutConfigLabel->text().trimmed();
342  if (currentHex.startsWith("0x", Qt::CaseInsensitive)) currentHex = currentHex.mid(2);
343 
344  InputDialog ipd("Change configuration string", "New configuration string", currentHex);
345  HexadecimalValidator hexValidator;
346  ipd.addValidator(&hexValidator);
347  if (ipd.exec() != QDialog::Accepted || ipd.textValue().isEmpty()) return;
348 
349  // the slice of the selected pin has a fixed number of digits, shorter input is padded with leading zeros
350  const int numDigits = (bitCount + 3) / 4;
351  QString sliceHex = ipd.textValue().toUpper();
352  if (sliceHex.size() > numDigits)
353  {

355  return;
356  }
357  sliceHex = sliceHex.rightJustified(numDigits, '0');
358 
359  auto initHex = getLutInitHex(mCurrentGate, initKey);
360  if (initHex.is_error())
361  {
362  log_error("gui", "Could not load INIT parameter from gate with id {}: {}", mCurrentGate->get_id(), initHex.get_error().get());
363  return;
364  }
365  auto spliceRes = LUTComponent::splice_init_slice(initHex.get(), sliceHex.toStdString(), bitOffset, bitCount);
366  if (spliceRes.is_error())
367  {
368  log_error("gui", "Could not change configuration string of gate with id {}: {}", mCurrentGate->get_id(), spliceRes.get_error().get());
369  return;
370  }
371 
372  ActionSetParameterValue* act = new ActionSetParameterValue(QString::fromStdString(initKey), "0x" + QString::fromStdString(spliceRes.get()));
373  act->setObject(UserActionObject(mCurrentGate->get_id(), UserActionObjectType::Gate));
374  act->exec();
375  setGate(mCurrentGate); // must update config string and parameter table, no signal for that
376  });

378  QApplication::clipboard()->setText(PyCodeProvider::pyCodeGateParameterValue(mCurrentGate->get_id(), QString::fromStdString(initKey)));
379  });
