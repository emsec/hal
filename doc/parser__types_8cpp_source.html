186  Result<std::vector<Token>> reverse_polish_notation(std::vector<Token>&& tokens, const std::string& expression, const ParserType& parser)
187  {
188  // Validate ternary structure: every `:` must be preceded (in source order) by an
189  // unmatched `?`, and every `?` must eventually have a matching `:`. This catches
190  // isolated `?`/`:`, wrong order (e.g. `a : b ? c`), and unbalanced/incomplete
191  // ternaries before the Shunting Yard step silently accepts them.
192  int ternary_depth = 0;
193  for (const auto& tok : tokens)
194  {
195  if (tok.is(TokenType::Question))
196  {
197  ++ternary_depth;
198  }
199  else if (tok.is(TokenType::Colon))
200  {
201  if (ternary_depth == 0)
202  {
203  return ERR("could not translate '" + expression + "' to reverse polish notation: unmatched ':' in ternary expression");
204  }
205  --ternary_depth;
206  }
207  }
208  if (ternary_depth != 0)
209  {
210  return ERR("could not translate '" + expression + "' to reverse polish notation: unmatched '?' in ternary expression");
211  }
212 
213  std::stack<Token> operator_stack;
214  std::vector<Token> output;
215 
216  for (auto&& token : tokens)
217  {
218  switch (token.type)
219  {
220  // (1) constants / variables are always pushed into output queue
221  case TokenType::Constant:
222  case TokenType::Variable: {
223  output.emplace_back(token);
224  break;
225  }
226 
227  // (2) operations are pushed into the operator stack with respect
228  // to the operator precedence and bracket level
229  case TokenType::And:
230  case TokenType::Or:
231  case TokenType::Not:
232  case TokenType::Xor:
233  case TokenType::Eq:
234  case TokenType::Question:
235  case TokenType::Colon:
236  case TokenType::Add:
237  case TokenType::Sub:
238  case TokenType::Mul: {
239  const bool left_assoc = token.is_left_associative();
240  while (!operator_stack.empty() && !operator_stack.top().is(TokenType::BracketOpen)
241  && (operator_stack.top().precedence(parser) > token.precedence(parser)
242  || (left_assoc && operator_stack.top().precedence(parser) == token.precedence(parser))))
243  {
244  output.emplace_back(operator_stack.top());
245  operator_stack.pop();
246  }
247  operator_stack.push(token);
248  break;
249  }
250 
251  // (3) the liberty grammar contains suffix not operations that
252  // invert the previous operation, i.e, acts similiar to
253  // a "not" operation in reverse-polish notation. hence,
254  // we simply push a "not" the output as the previous
255  // expression has been already translated to the reverse-
256  // polish notation in the output vector.
257  case TokenType::NotSuffix: {
258  output.emplace_back(Token::Not());
259  break;
260  }
261 
262  // (4) open brackets are always pushed into the operator stack
263  case TokenType::BracketOpen: {
264  operator_stack.push(token);
265  break;
266  }
267 
268  // (5) close brackets push remaining operations into the output
269  // queue until the bracket level is closed
270  case TokenType::BracketClose: {
271  while (!operator_stack.empty() && !operator_stack.top().is(TokenType::BracketOpen))
272  {
273  output.emplace_back(operator_stack.top());
274  operator_stack.pop();
275  }
276 
277  operator_stack.pop();
278 
279  break;
280  }
281  }
282  }
283 
284  while (!operator_stack.empty())
285  {
286  if (operator_stack.top().is(TokenType::BracketOpen))
287  {
288  return ERR("could not translate '" + expression + "' to reverse polish notation: bracket level is invalid");
289  }
290 
291  output.emplace_back(operator_stack.top());
292  operator_stack.pop();
293  }
294  return OK(output);
295  }

298  {

303  {
304  if ((token.is(BooleanFunctionParser::TokenType::And) || token.is(BooleanFunctionParser::TokenType::Or) || token.is(BooleanFunctionParser::TokenType::Not) || token.is(BooleanFunctionParser::TokenType::Xor)
305  || token.is(BooleanFunctionParser::TokenType::Eq) || token.is(BooleanFunctionParser::TokenType::Colon)
306  || token.is(BooleanFunctionParser::TokenType::Add) || token.is(BooleanFunctionParser::TokenType::Sub) || token.is(BooleanFunctionParser::TokenType::Mul))
307  && nodes.empty())

310  }
311 
312  switch (token.type)

316  break;

318  nodes.emplace_back(BooleanFunction::Node::Operation(BooleanFunction::NodeType::Or, nodes.back().size));
319  break;
320  case BooleanFunctionParser::TokenType::Not:
321  nodes.emplace_back(BooleanFunction::Node::Operation(BooleanFunction::NodeType::Not, nodes.back().size));
322  break;
323  case BooleanFunctionParser::TokenType::Xor:
324  nodes.emplace_back(BooleanFunction::Node::Operation(BooleanFunction::NodeType::Xor, nodes.back().size));
325  break;
326  case BooleanFunctionParser::TokenType::Add:
327  nodes.emplace_back(BooleanFunction::Node::Operation(BooleanFunction::NodeType::Add, nodes.back().size));
328  break;
329  case BooleanFunctionParser::TokenType::Sub:
330  nodes.emplace_back(BooleanFunction::Node::Operation(BooleanFunction::NodeType::Sub, nodes.back().size));
331  break;
332  case BooleanFunctionParser::TokenType::Mul:
333  nodes.emplace_back(BooleanFunction::Node::Operation(BooleanFunction::NodeType::Mul, nodes.back().size));
334  break;
335  case BooleanFunctionParser::TokenType::Eq:
336  // Eq returns a 1-bit boolean result regardless of operand width.
337  nodes.emplace_back(BooleanFunction::Node::Operation(BooleanFunction::NodeType::Eq, 1));

341  break;

344  // The branches are the two top-most nodes (then below else on the stack).
345  if (nodes.size() < 3)
346  {

348  }
349  const u16 ite_size = nodes.back().size;
350  nodes.emplace_back(BooleanFunction::Node::Operation(BooleanFunction::NodeType::Ite, ite_size));
351  break;
352  }
353 
354  case BooleanFunctionParser::TokenType::Variable:
355  nodes.emplace_back(BooleanFunction::Node::Variable(std::get<0>(token.variable), std::get<1>(token.variable)));
356  break;
357  case BooleanFunctionParser::TokenType::Constant:
358  nodes.emplace_back(BooleanFunction::Node::Constant(token.constant));
359  break;
360 
361  default:
362  return ERR("could not translate tokens: unable to handle '" + token.to_string() + "' in '" + expression + "'");
363  }
364  }
365

367  {
368  return ERR_APPEND(res.get_error(), "could not translate tokens: unable to build Boolean function from vector of nodes");
369  }
370  else
371  {
372  return res;
373  }
