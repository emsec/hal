291  void WaveGraphicsCanvas::toggleZoom()
292  {
293  // try zoom out
294  int w = viewport()->size().width();
295  double dt = mTransform.deltaT();
296  if (dt <= 0) return;
297  double newScale = w / dt;
298  double vleft = 0;
299  if (dt * mTransform.scale() * 1.0001 < w)
300  {
301  // not using entire screen width, zoom in
302  mTransform.setScale(newScale);
303  }
304  else if (mTransform.scale() <= newScale)
305  {
306  // already zoomed out to max, try zoom in
307  TimeInterval shortest;
308  for (const WaveItem* wi : mWaveItemHash->values())
309  {
310  if (!wi->isVisibile() || ! wi->mVisibleRange) continue;
311  if (wi->mPainted.shortestToggle() < shortest) shortest = wi->mPainted.shortestToggle();
312  }
313  if (shortest.mCenterTime <= 0 || shortest.mDuration <= 0) return;
314  newScale = 6 / shortest.mDuration;
315  if (mTransform.scale() >= newScale) return;
316  mTransform.setScale(newScale);
317  vleft = mTransform.vPos(shortest.mCenterTime) - w/2;
318  if (vleft + w > mTransform.vMax()) vleft = mTransform.vMax() - w;
319  if (vleft < 0) vleft = 0;
320  }
321  else
322  {
323  // zoom out
324  mTransform.setScale(newScale);
325  }
326  mScrollbar->setVleft(vleft);
327  mRenderEngine->update();
328  mTimescale->setScale(viewport()->size().width());
329  mCursor->recalcTime();
330  }
