if(BUILD_DOCUMENTATION)
    find_package(Doxygen REQUIRED OPTIONAL_COMPONENTS dot)

    # Graphs are only generated if graphviz is actually available, otherwise doxygen
    # aborts with "Problems running dot" for every single graph.
    if(TARGET Doxygen::dot)
        set(DOXYGEN_HAVE_DOT YES)
        get_target_property(DOXYGEN_DOT_EXECUTABLE Doxygen::dot IMPORTED_LOCATION)
        get_filename_component(DOXYGEN_DOT_PATH ${DOXYGEN_DOT_EXECUTABLE} DIRECTORY)
    else()
        message(STATUS "graphviz (dot) not found, doxygen graphs will be disabled")
        set(DOXYGEN_HAVE_DOT NO)
        set(DOXYGEN_DOT_PATH)
    endif()

    set(DOXYGEN_IN ${CMAKE_SOURCE_DIR}/documentation/Doxyfile.in)
    set(DOXYGEN_OUT ${CMAKE_BINARY_DIR}/documentation/Doxyfile)

    configure_file(${DOXYGEN_IN} ${DOXYGEN_OUT} @ONLY)
    message("Building Doxygen documentation...")

    add_custom_target(doc ALL
                      COMMAND ${DOXYGEN_EXECUTABLE} ${DOXYGEN_OUT}
                      WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
                      COMMENT "Generating API documentation with Doxygen"
                      VERBATIM )


    # The python documentation is generated by importing the freshly built hal_py
    # extension module. That module is linked against ${Python3_EXECUTABLE}, so sphinx
    # has to run on exactly that interpreter -- a sphinx-build found somewhere on PATH
    # may well belong to a different python version and would crash on import.
    if(NOT Python3_EXECUTABLE)
        message(FATAL_ERROR "BUILD_DOCUMENTATION requires a python 3 interpreter, but none was found.")
    endif()

    # Note: the hint below deliberately mentions the virtual environment route as well.
    # Distribution-provided interpreters (Homebrew, Debian, ...) are usually marked as
    # "externally managed" (PEP 668), which makes a plain 'pip install' fail. Beware that
    # the interpreter reported here is often NOT the 'python3' first on PATH.
    set(SPHINX_INSTALL_HINT
        "Either install it into that interpreter\n"
        "    ${Python3_EXECUTABLE} -m pip install sphinx sphinx_rtd_theme\n"
        "  or, if that interpreter is externally managed (PEP 668), use a virtual environment\n"
        "    ${Python3_EXECUTABLE} -m venv <venv-dir>\n"
        "    <venv-dir>/bin/pip install sphinx sphinx_rtd_theme\n"
        "    cmake -DPython3_EXECUTABLE=<venv-dir>/bin/python <...>")

    execute_process(COMMAND ${Python3_EXECUTABLE} -c "import sphinx; print(sphinx.__version__)"
                    RESULT_VARIABLE SPHINX_IMPORT_RESULT
                    OUTPUT_VARIABLE SPHINX_VERSION
                    OUTPUT_STRIP_TRAILING_WHITESPACE
                    ERROR_QUIET)
    if(NOT SPHINX_IMPORT_RESULT EQUAL 0)
        message(FATAL_ERROR
                "BUILD_DOCUMENTATION requires 'sphinx' to be importable from\n"
                "    ${Python3_EXECUTABLE}\n"
                "  which is the interpreter the hal_py extension module is built against.\n"
                "  " ${SPHINX_INSTALL_HINT})
    endif()
    message(STATUS "Found Sphinx ${SPHINX_VERSION} for ${Python3_EXECUTABLE}")

    execute_process(COMMAND ${Python3_EXECUTABLE} -c "import sphinx_rtd_theme"
                    RESULT_VARIABLE SPHINX_RTD_THEME_RESULT
                    ERROR_QUIET)
    if(NOT SPHINX_RTD_THEME_RESULT EQUAL 0)
        message(FATAL_ERROR
                "The python documentation uses the 'sphinx_rtd_theme' html theme, but it is not "
                "importable from\n"
                "    ${Python3_EXECUTABLE}\n"
                "  " ${SPHINX_INSTALL_HINT})
    endif()

    # configured documentation tools and intermediate build results
    set(BINARY_BUILD_DIR "${CMAKE_BINARY_DIR}/documentation/python-doc/_build")

    # Sphinx cache with pickled ReST documents
    set(SPHINX_CACHE_DIR "${CMAKE_BINARY_DIR}/documentation/python-doc/_doctrees")

    # HTML output directory
    set(SPHINX_HTML_DIR "${CMAKE_BINARY_DIR}/documentation/python-doc/html")

    configure_file(
            "sphinx_doc/conf.py.in"
            "${BINARY_BUILD_DIR}/conf.py"
            @ONLY)

    file(GLOB_RECURSE SPHINX_DOC_FILES ${CMAKE_CURRENT_SOURCE_DIR}/sphinx_doc/*.rst)

    file(MAKE_DIRECTORY ${SPHINX_BUILD_DIR})
    message(STATUS "SPHINX_DOC_FILES: ${SPHINX_DOC_FILES}")

    foreach(file ${SPHINX_DOC_FILES})
        get_filename_component(f ${file} NAME)
        list(APPEND DESTINATION_DOC_FILES ${SPHINX_BUILD_DIR}/${f})
    endforeach()

    get_property(SPHINX_PLUGIN_TARGETS_LOCAL GLOBAL PROPERTY SPHINX_PLUGIN_TARGETS)

    foreach(idx ${SPHINX_PLUGIN_TARGETS_LOCAL})
        set(PYDOC_PLUGINS "${PYDOC_PLUGINS}\n   ${idx}")
    endforeach()

    configure_file(${CMAKE_CURRENT_SOURCE_DIR}/sphinx_doc/plugins.rst.in ${SPHINX_BUILD_DIR}/plugins.rst @ONLY)

    # Note: pass the file list explicitly instead of a '*.rst' glob -- 'cmake -E copy' does
    # not expand wildcards itself, it only works by accident on generators that route the
    # command through a shell.
    add_custom_target(copy_sphinx_doc_files ALL
                      BYPRODUCTS ${DESTINATION_DOC_FILES}
                      COMMAND ${CMAKE_COMMAND} -E copy_if_different ${SPHINX_DOC_FILES} ${SPHINX_BUILD_DIR}/
                      DEPENDS ${SPHINX_DOC_FILES}
                      WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
                      COMMENT "Update sphinx doc files")

    add_custom_target(pydoc ALL
                      ${Python3_EXECUTABLE} -m sphinx
                      -b html
                      -c "${BINARY_BUILD_DIR}"
                      -d "${SPHINX_CACHE_DIR}"
                      "${SPHINX_BUILD_DIR}"
                      "${SPHINX_HTML_DIR}"
                      COMMENT "Building HTML documentation with Sphinx" VERBATIM
                      DEPENDS hal::py copy_sphinx_doc_files)

    add_dependencies(pydoc ${SPHINX_PLUGIN_TARGETS_LOCAL})

endif()
